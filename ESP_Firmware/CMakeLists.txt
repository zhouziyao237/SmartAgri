cmake_minimum_required(VERSION 3.16)
if(ESP_IDF_SDK_PATH)
  file(TO_CMAKE_PATH "${ESP_IDF_SDK_PATH}" _idf)
  set(ENV{IDF_PATH} "${_idf}")
endif()
if(ESP_IDF_TOOLS_PATH)
  file(TO_CMAKE_PATH "${ESP_IDF_TOOLS_PATH}" _tools)
  set(ENV{IDF_TOOLS_PATH} "${_tools}")
endif()
# IDF's ROM gdbinit generator reads an ENVIRONMENT variable, not a CMake cache variable.
if(ESP_IDF_ROM_ELF_DIR)
  file(TO_CMAKE_PATH "${ESP_IDF_ROM_ELF_DIR}" _rom)
  set(ENV{ESP_ROM_ELF_DIR} "${_rom}")
endif()
if(ESP_IDF_TOOLCHAIN_BIN)
  file(TO_CMAKE_PATH "${ESP_IDF_TOOLCHAIN_BIN}" _tcbin)
  if(WIN32)
    set(ENV{PATH} "${_tcbin};$ENV{PATH}")
  else()
    set(ENV{PATH} "${_tcbin}:$ENV{PATH}")
  endif()
endif()
if(ESP_IDF_PYTHON_EXE)
  set(PYTHON "${ESP_IDF_PYTHON_EXE}" CACHE FILEPATH "" FORCE)
endif()
if(NOT DEFINED ENV{IDF_PATH})
  message(FATAL_ERROR "Set ESP_IDF_SDK_PATH or IDF_PATH")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(smartagri_esp_dualtarget_cluster_v2_4_8)
